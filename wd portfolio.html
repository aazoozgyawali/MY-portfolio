Demo 2: Nepal Income Tax Calculator (C)

A C program calculating annual personal income tax based on Nepal's current individual tax ceiling.

 
#include 

int main() {
    float income, tax = 0;
    
    printf("Enter Annual Income: ");
    scanf("%f", &income);
    
    if (income <= 500000) {
        tax = income * 0.01;
    } else if (income <= 700000) {
        tax = 5000 + (income - 500000) * 0.10;
    } else if (income <= 1000000) {
        tax = 25000 + (income - 700000) * 0.20;
    } else {
        tax = 85000 + (income - 1000000) * 0.30;
    }
    
    printf("Total Tax = %.2f\n", tax);
    
    return 0;
}
